    print("Score:       ", f"{score * 100:.2f}%" if score <= 1 else f"{score:.2f}%")
else:
    print("Score:       ", score)

print("=" * 60)

for case in r.get("cases", []):
    actual = case.get("actualStatus") or case.get("actual")
    expected = case.get("expectedStatus") or case.get("expected")
    ok = case.get("passed")
    print(
        f"{'PASS' if ok else 'FAIL':4}  "
        f"{case.get('id', '?'):40} "
        f"expected={expected} actual={actual}"
    )

cd /home/projects/Cortexo && python3 - <<'PY'
import json
from pathlib import Path

p = Path("artifacts/evaluations/regression/latest.json")
if not p.exists():
    raise SystemExit(f"ERROR: {p} not found")

r = json.loads(p.read_text(encoding="utf-8"))

print("=" * 100)
print("DETERMINISTIC HARNESS ERRORS")
print("=" * 100)

cases = (
    r.get("deterministicCases")
    or r.get("deterministic_cases")
    or r.get("cases")
    or r.get("deterministic", {}).get("cases")
    or []
)

for case in cases:
    actual = (
        case.get("actualStatus")
        or case.get("actual_status")
        or case.get("actual")
        or case.get("status")
    )

    if actual != "HARNESS_ERROR":
        continue

    print()
    print("CASE:", case.get("id") or case.get("caseId"))
    print("EXPECTED:", case.get("expectedStatus") or case.get("expected_status") or case.get("expected"))

    for key in (
        "error",
        "errorMessage",
        "error_message",
        "exception",
        "message",
        "stderr",
        "details",
        "graderResult",
        "grader_result",
    ):
        if key in case and case[key] not in (None, "", {}, []):
            print(f"{key}:")
            value = case[key]
            if isinstance(value, (dict, list)):
                print(json.dumps(value, indent=2))
            else:
                print(value)

    print("-" * 100)
PY
